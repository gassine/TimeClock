
// ---------------------------------------------------------
// TRAINING MODULE MODELS
// ---------------------------------------------------------

model TrainingCategory {
  id              String         @id @default(cuid())
  name            String
  description     String?
  isActive        Boolean        @default(true)
  isDeleted       Boolean        @default(false)
  order           Int            @default(0)
  
  // Permissions
  isAdminOnly     Boolean        @default(false)
  isEveryone      Boolean        @default(true) // If true, ignores role restrictions
  viewRoles       String         @default("[]") // JSON Array of Role IDs allowed to view
  postRoles       String         @default("[]") // JSON Array of Role IDs allowed to post
  
  // Relations
  posts           TrainingPost[]
  
  createdAt       DateTime       @default(now())
  updatedAt       DateTime       @updatedAt
}

model TrainingPost {
  id              String         @id @default(cuid())
  title           String
  content         String         // Raw text or Sanitized HTML
  status          String         @default("ACTIVE") // DRAFT, ACTIVE, ARCHIVED
  isPinned        Boolean        @default(false)
  isDeleted       Boolean        @default(false)
  allowReplies    Boolean        @default(true)
  order           Int            @default(0)
  expiresAt       DateTime?      // Auto-archive or hide after this date
  
  // Relations
  categoryId      String
  category        TrainingCategory @relation(fields: [categoryId], references: [id])
  authorId        String
  author          Firefighter    @relation(fields: [authorId], references: [id])
  
  replies         TrainingReply[]
  readStatuses    TrainingReadStatus[]
  versions        TrainingPostVersion[]
  
  createdAt       DateTime       @default(now())
  updatedAt       DateTime       @updatedAt
}

model TrainingReply {
  id              String         @id @default(cuid())
  content         String
  isDeleted       Boolean        @default(false)
  
  // Relations
  postId          String
  post            TrainingPost   @relation(fields: [postId], references: [id])
  authorId        String
  author          Firefighter    @relation(fields: [authorId], references: [id])
  
  createdAt       DateTime       @default(now())
  updatedAt       DateTime       @updatedAt
}

model TrainingReadStatus {
  id              String         @id @default(cuid())
  
  // Relations
  postId          String
  post            TrainingPost   @relation(fields: [postId], references: [id])
  userId          String
  user            Firefighter    @relation(fields: [userId], references: [id])
  
  readAt          DateTime       @default(now())

  @@unique([postId, userId])
}

model TrainingPostVersion {
  id              String         @id @default(cuid())
  title           String
  content         String
  
  // Relations
  postId          String
  post            TrainingPost   @relation(fields: [postId], references: [id])
  editorId        String?        // ID of the user who made the edit
  
  createdAt       DateTime       @default(now())
}
